  function purchaseDomain(domain, source, chk) {
    var form = chk ? document.purchaseFormChk : document.purchaseForm;
    var submitted = false;
    var submitForm = function(){if(!submitted){submitted=true;form.submit();}}

  }

  async function refreshDomain(domain) {
    const refreshEl = document.getElementById('refresh');
    if(refreshEl.classList.contains('spin')) return;
  
    refreshEl.classList.add('spin');
    const statusEl = document.querySelector('#refresh .status');
    statusEl.innerHTML = '';
    
    const response = await fetch(`/whois/refresh.php?domain=${domain}`, {headers:{"X-Requested-With":"XMLHttpRequest"}})
    if(response.ok) {
      const data = await response.json();
      let status = 'Error occurred. Please try again later.';
      if(data.errno == 0) {
        if(data.status == 1) {
          let el;
          document.querySelector('.whois-data').style.display = 'none';
          if((el = document.querySelector('.section-related'))) {
            el.style.display = 'none';
          }
          if((el = document.querySelector('.section-avail'))) {
            el.innerHTML = `<table><tr><td><img src="/images/ok.png" alt="ok"></td><td>${domain} is available!</td></tr></table>`;
          }
        } else {
          document.querySelectorAll('.df-block').forEach((el) => el.remove());
          document.querySelectorAll('.df-block-raw').forEach((el) => el.remove());
          document.querySelector('.whois-data .head').insertAdjacentHTML('afterend', data['blocks']);
          document.querySelector('#refresh .age').innerHTML = 'Updated '+data['dataAge']+' ago';
          history.replaceState(null, '');
        }
        status = '';
      } else if(data.captcha) {
        status = '';
        document.getElementById('security').style.display = '';
        document.querySelector('#security form').insertAdjacentHTML('afterbegin', '<altcha-widget id="altcha" challengeurl="/whois/altcha.php" hidefooter="true"></altcha-widget>');
        document.head.appendChild(Object.assign(document.createElement('script'), {src:'https://fastly.jsdelivr.net/npm/altcha@2.2.4/dist/altcha.js', type:'module', async:true}));
      } else if(data.errstr) {
        status = data.errstr;
      }
      refreshEl.classList.remove('spin')
      statusEl.innerHTML = status;
    } else {
      refreshEl.classList.remove('spin')
      statusEl.innerHTML = 'Error encountered. Please try again later.';
    }
  }
  
  function refreshWhois() {

  async function suggestDomain(domain) {
    const response = await fetch(`/whois/suggest.php?domain=${domain}`, {headers:{"X-Requested-With":"XMLHttpRequest"}})
    if(response.ok) {
      const data = await response.json();
      let html = '';
      for(const value of Object.values(data)) {
        html += `<div class="item"><span>${value.name}</span><a href="javascript:purchaseDomain('${value.name}','suggest');" class="btn btn-sm btn-vendor">Buy Now</a></div>`;
      }
      if(html) {
        document.querySelector('aside').insertAdjacentHTML('afterbegin', '<div class="chiclet-suggest"><div class="title">Interested in similar domains?</div>'+html+'</div>');
      }
    }
  }
